# the whole app, in short (Stage 5's server.py, without the ? parameter)
from bottle import route, run
import sqlite3

@route('/students')
def list_students():
    con = sqlite3.connect('team_details.db')
    rows = con.execute(
        'SELECT student_name, inter_college, inter_city FROM students'
    ).fetchall()
    con.close()
    result = []
    for row in rows:
        name, college, city = row
        result.append({'student_name': name, 'inter_college': college, 'inter_city': city})
    return {'students': result}

run(host='localhost', port=8080)
